def upload_to_github(content, film_title, message="Upload M3U file"):
    """
    Fungsi untuk mengunggah file ke GitHub jika judul film belum ada.
    Jika file sudah ada, menambahkan baris baru ke dalam file dan menambahkan jarak antar baris.
    """
    try:
        print("\nMengunggah file ke GitHub...")
        url = f"https://api.github.com/repos/{GITHUB_REPO}/contents/{GITHUB_FILE_PATH}"
        headers = {
            "Authorization": f"token {GITHUB_TOKEN}",
            "Accept": "application/vnd.github+json"
        }

        # Mengecek apakah file sudah ada di repositori
        response = requests.get(url, headers=headers)
        if response.status_code == 200:
            file_content = base64.b64decode(response.json()["content"]).decode()
            if film_title.lower() in file_content.lower():
                print(f"Film '{film_title}' sudah ada di file GitHub. Tidak menambahkan lagi.")
                return  # Tidak melanjutkan jika film sudah ada
            sha = response.json()["sha"]  # SHA file untuk overwrite
            # Menambahkan baris kosong sebagai jarak antar film
            content = file_content + "\n" + content  # Menambahkan baris kosong sebelum menambahkan konten baru
        else:
            sha = None
            content = content  # Jika file tidak ada, buat file baru

        # Membuat payload untuk unggah file
        payload = {
            "message": message,
            "content": base64.b64encode(content.encode()).decode(),
        }
        if sha:
            payload["sha"] = sha

        # Mengunggah file
        response = requests.put(url, headers=headers, json=payload)
        if response.status_code in [200, 201]:
            print("File berhasil diunggah ke GitHub.")
            print(f"URL File: {response.json()['content']['html_url']}")
        else:
            print(f"Gagal mengunggah file: {response.json()}")
    except Exception as e:
        print(f"Terjadi kesalahan saat mengunggah ke GitHub: {e}")

def get_playeriframe_link(url):
    try:
        print(f"Mengambil data dari: {url}...")
        # Mengambil konten halaman dengan cloudscraper
        response = scraper.get(url)
        response.raise_for_status()  # Memastikan permintaan berhasil
        html_content = response.text

        # Mem-parsing HTML menggunakan BeautifulSoup
        soup = BeautifulSoup(html_content, "html.parser")

        # Mencari semua link yang mengarah ke playeriframe.lol
        links = soup.find_all("a", href=True)
        playeriframe_links = [link['href'] for link in links if "playeriframe.lol" in link['href']]

        # Menampilkan semua link yang ditemukan
        if playeriframe_links:
            print("\nLink playeriframe.lol yang ditemukan:")
            for i, link in enumerate(playeriframe_links, 1):
                print(f"{i}. {link}")
            
            # Mencari link yang mengandung "emturbovid.com"
            specific_link = next((link for link in playeriframe_links if "emturbovid.com" in link), None)
            if specific_link:
                print("\nLink spesifik ditemukan:")
                print(specific_link)

                # Mengubah link menjadi link langsung ke emturbovid.com
                parsed_url = urllib.parse.urlparse(specific_link)
                query_params = urllib.parse.parse_qs(parsed_url.query)
                if "url" in query_params:
                    direct_link = urllib.parse.unquote(query_params["url"][0])
                    print("\nLink yang diubah menjadi: ")
                    print(direct_link)

                    # Mencari logo film dari halaman awal
                    logo_url = find_logo(soup, url)

                    # Mendapatkan genre film
                    genre = get_genre(soup)

                    # Mencari file m3u8 dalam halaman langsung
                    find_m3u8(direct_link, url, logo_url, genre)
                    return direct_link
                else:
                    print("\nGagal mengubah link spesifik.")
                    return None
            else:
                print("\nTidak ada link spesifik dengan 'emturbovid.com' ditemukan.")
                return None
        else:
            print("Tidak ada link playeriframe.lol yang ditemukan di halaman ini.")
            return None
    except requests.exceptions.RequestException as e:
        print(f"Terjadi kesalahan saat mengambil data: {e}")

def get_genre(soup):
    """
    Mengambil genre film dari halaman web.
    """
    print("\nMencari genre film...")
    try:
        # Mencari elemen <h2> yang mengandung teks "Genre"
        genre_section = soup.find("h2", string=lambda text: text and "Genre" in text)
        if genre_section:
            # Mencari elemen berikutnya yang berisi daftar genre
            genre_links = genre_section.find_next("h3").find_all("a")
            if genre_links:
                # Ambil genre pertama
                genre = genre_links[0].text.strip()
                # Terjemahkan genre ke bahasa Indonesia jika tersedia dalam dictionary
                genre_translated = GENRE_TRANSLATION.get(genre, genre)
                print(f"Genre ditemukan: {genre} (diterjemahkan: {genre_translated})")
                return genre_translated
    except Exception as e:
        print(f"Kesalahan saat mencari genre: {e}")
    
    # Jika genre tidak ditemukan
    print("Genre tidak ditemukan. Menggunakan default: 'Film'.")
    return "Film"

def find_logo(soup, page_url):
    """
    Mencari URL logo film dari halaman.
    """
    print("\nMencari logo film...")
    film_name = page_url.split("/")[-2]
    img_tags = soup.find_all("img", src=True)

    for img in img_tags:
        src = img["src"]
        if film_name in src and "film" in src:
            print(f"Logo film ditemukan: {src}")
            return src
    print("Logo film tidak ditemukan, menggunakan logo default.")
    return "https://s7.lk21static.buzz/wp-content/uploads/2024/11/default-film-logo.jpg"

def find_m3u8(direct_link, original_url, logo_url, genre):
    try:
        print(f"\nMengakses halaman: {direct_link} untuk mencari file m3u8...")
        response = scraper.get(direct_link)
        response.raise_for_status()
        html_content = response.text

        if ".m3u8" in html_content:
            m3u8_links = [line for line in html_content.splitlines() if ".m3u8" in line]
            if m3u8_links:
                m3u8_url = m3u8_links[0]  # Ambil m3u8 pertama yang ditemukan
                print(f"\nMenemukan link m3u8: {m3u8_url}")

                # Ambil judul film dari URL
                film_title = original_url.split("/")[-2].replace("-", " ").title()

                # Membuat format M3U
                m3u_content = create_m3u_format(m3u8_url, original_url, genre, logo_url)
                upload_to_github(m3u_content, film_title)
    except requests.exceptions.RequestException as e:
        print(f"Kesalahan saat mengambil data m3u8: {e}")

def create_m3u_format(m3u8_url, original_url, genre, logo_url):
    """
    Fungsi ini membuat format M3U yang menyertakan URL gambar pada `tvg-logo`.
    """
    # Ambil nama film dari URL dan ubah ke format title case
    film_title_raw = original_url.split("/")[-2].replace("-", " ").title()
    
    # Deteksi dan tambahkan tahun ke dalam tanda kurung
    parts = film_title_raw.split()
    if parts[-1].isdigit() and len(parts[-1]) == 4:  # Jika bagian terakhir adalah tahun (4 digit)
        film_title = " ".join(parts[:-1]) + f" ({parts[-1]})"
    else:
        film_title = film_title_raw  # Jika tidak ada tahun, gunakan format biasa

    # Membuat konten M3U dengan menambahkan `tvg-logo`
    m3u_content = (
        f"#EXTINF:-1 tvg-logo=\"{logo_url}\" group-title=\"{genre}\",{film_title}\n"
        f"{m3u8_url}\n"  # Menambahkan satu baris kosong setelah setiap entri
    )
    
    print("\nKonten M3U yang dibuat:")
    print(m3u_content)
    return m3u_content
